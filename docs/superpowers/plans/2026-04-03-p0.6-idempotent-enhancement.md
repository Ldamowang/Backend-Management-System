```java
@ExceptionHandler(BusinessException.class)
public ResponseEntity<Result<Void>> handleBusinessException(BusinessException e) {
    int code = e.getCode();
    Result<Void> result = Result.fail(code, e.getMessage());
    // 409 冲突需要返回 HTTP 409 状态码
    if (code == 409) {
        return ResponseEntity.status(HttpStatus.CONFLICT).body(result);
    }
    return ResponseEntity.ok(result);
}
```

## Task 3: 前端 useSubmit Composable

**Files:**
- Create: `admin-frontend/src/composables/useSubmit.ts`

- [ ] **Step 1: 创建 useSubmit**

```typescript
import { ref } from 'vue'
import { ElMessage } from 'element-plus'

/**
 * 统一表单提交防重 composable
 *
 * 功能：
 * - 自动管理 loading 状态
 * - 节流控制（默认 300ms）
 * - 成功后自动重置状态
 * - 可选成功提示
 *
 * 用法：
 * const { submitting, submit } = useSubmit()
 * submit(() => api.createUser(form), {
 *   successMessage: '创建成功',
 *   onSuccess: () => fetchData()
 * })
 */
export function useSubmit() {
  const submitting = ref(false)
  let lastSubmitTime = 0
  const THROTTLE_MS = 300

  const submit = async <T>(
    apiFn: () => Promise<T>,
    options?: {
      successMessage?: string
      onSuccess?: (result: T) => void
      onError?: (error: Error) => void
      throttle?: number
    }
  ): Promise<T | undefined> => {
    // 节流控制
    const now = Date.now()
    const throttle = options?.throttle ?? THROTTLE_MS
    if (now - lastSubmitTime < throttle) {
      return undefined
    }
    lastSubmitTime = now

    // 防止重复提交
    if (submitting.value) {
      return undefined
    }

    submitting.value = true
    try {
      const result = await apiFn()
      if (options?.successMessage) {
        ElMessage.success(options.successMessage)
      }
      options?.onSuccess?.(result)
      return result
    } catch (error: any) {
      if (options?.onError) {
        options.onError(error)
      } else {
        ElMessage.error(error.message || '操作失败')
      }
      return undefined
    } finally {
      submitting.value = false
    }
  }

  return { submitting, submit }
}
```
