ON DUPLICATE KEY UPDATE config_key = config_key;

```java
package com.iflytek.admin.modules.auth.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.iflytek.admin.common.exception.BusinessException;
import com.iflytek.admin.common.result.ResultCode;
import com.iflytek.admin.modules.auth.dto.TotpSetupResponse;
import com.iflytek.admin.modules.system.entity.SysUser2fa;
import com.iflytek.admin.modules.system.mapper.SysUser2faMapper;
import com.warrenstrange.googleauth.GoogleAuthenticator;
import com.warrenstrange.googleauth.GoogleAuthenticatorKey;
import com.warrenstrange.googleauth.GoogleAuthenticatorQRGenerator;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.security.SecureRandom;
import java.util.ArrayList;
import java.util.List;

@Service
@RequiredArgsConstructor
public class TotpService {

    private final SysUser2faMapper user2faMapper;
    private final ObjectMapper objectMapper;
    private static final String ISSUER = "AdminSystem";
    private static final GoogleAuthenticator gAuth = new GoogleAuthenticator();

    /**
     * 生成 2FA 密钥和二维码，返回给前端展示
     */
    public TotpSetupResponse generateSetup(Long userId, String username) {
        // 如果已有记录且已启用，不允许重复绑定
        SysUser2fa existing = getByUserId(userId);
        if (existing != null && existing.getEnabled() == 1) {
            throw new BusinessException(ResultCode.BAD_REQUEST.getCode(), "2FA已启用，请先禁用后重新绑定");
        }

        GoogleAuthenticatorKey key = gAuth.createCredentials();
        String secret = key.getKey();
        String qrCodeUri = GoogleAuthenticatorQRGenerator.getOtpAuthTotpURL(ISSUER, username, key);
        List<String> backupCodes = generateBackupCodes(10);

        // 保存或更新（未启用状态）
        if (existing != null) {
            existing.setSecretKey(secret);
            existing.setBackupCodes(toJson(backupCodes));
            existing.setEnabled(0);
            user2faMapper.updateById(existing);
        } else {
            SysUser2fa record = new SysUser2fa();
            record.setUserId(userId);
            record.setSecretKey(secret);
            record.setBackupCodes(toJson(backupCodes));
            record.setEnabled(0);
            user2faMapper.insert(record);
        }

        return TotpSetupResponse.builder()
                .secretKey(secret)
                .qrCodeUri(qrCodeUri)
                .backupCodes(backupCodes)
                .build();
    }

    /**
     * 验证 TOTP 码并启用 2FA
     */
    public void verifyAndEnable(Long userId, String code) {
        SysUser2fa record = getByUserId(userId);
        if (record == null) {
            throw new BusinessException(ResultCode.BAD_REQUEST.getCode(), "请先生成2FA密钥");
        }
        if (!gAuth.authorize(record.getSecretKey(), Integer.parseInt(code))) {
            throw new BusinessException(ResultCode.BAD_REQUEST.getCode(), "验证码错误，请重试");
        }
        record.setEnabled(1);
        user2faMapper.updateById(record);
    }

    /**
     * 禁用 2FA
     */
    public void disable(Long userId) {
        SysUser2fa record = getByUserId(userId);
        if (record != null) {
            user2faMapper.deleteById(record.getId());
        }
    }

    /**
     * 登录时校验 2FA（如果用户已启用）
     * 返回 true 表示通过或无需校验
     */
    public boolean validateForLogin(Long userId, String totpCode) {
        SysUser2fa record = getByUserId(userId);
        if (record == null || record.getEnabled() != 1) {
            return true; // 未启用 2FA，跳过
        }
        if (totpCode == null || totpCode.isBlank()) {
            return false; // 已启用但未提供验证码
        }
        // 先尝试 TOTP 验证
        try {
            if (gAuth.authorize(record.getSecretKey(), Integer.parseInt(totpCode))) {
                return true;
            }
        } catch (NumberFormatException ignored) {
            // 可能是备用码
        }
        // 再尝试备用码验证
        return validateBackupCode(record, totpCode);
    }

    /**
     * 检查用户是否已启用 2FA
     */
    public boolean isEnabled(Long userId) {
        SysUser2fa record = getByUserId(userId);
        return record != null && record.getEnabled() == 1;
    }

    private boolean validateBackupCode(SysUser2fa record, String code) {
        List<String> codes = fromJson(record.getBackupCodes());
        if (codes != null && codes.remove(code)) {
            record.setBackupCodes(toJson(codes));
            user2faMapper.updateById(record);
            return true;
        }
        return false;
    }

    private SysUser2fa getByUserId(Long userId) {
        return user2faMapper.selectOne(
                new LambdaQueryWrapper<SysUser2fa>().eq(SysUser2fa::getUserId, userId)
        );
    }

    private List<String> generateBackupCodes(int count) {
        SecureRandom random = new SecureRandom();
        List<String> codes = new ArrayList<>(count);
        for (int i = 0; i < count; i++) {
            codes.add(String.format("%08d", random.nextInt(100000000)));
        }
        return codes;
    }

    private String toJson(List<String> list) {
        try {
            return objectMapper.writeValueAsString(list);
        } catch (JsonProcessingException e) {
            throw new RuntimeException("JSON序列化失败", e);
        }
    }

    private List<String> fromJson(String json) {
        if (json == null || json.isBlank()) return new ArrayList<>();
        try {
            return objectMapper.readValue(json, new TypeReference<>() {});
        } catch (JsonProcessingException e) {
            return new ArrayList<>();
        }
    }
}
```

- [ ] **Step 1: 创建 TotpController**

```java
package com.iflytek.admin.modules.auth.controller;

import com.iflytek.admin.common.result.Result;
import com.iflytek.admin.common.utils.SecurityUtil;
import com.iflytek.admin.modules.auth.dto.TotpSetupResponse;
import com.iflytek.admin.modules.auth.dto.TotpVerifyRequest;
import com.iflytek.admin.modules.auth.service.TotpService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

@Tag(name = "双因素认证")
@RestController
@RequestMapping("/api/2fa")
@RequiredArgsConstructor
public class TotpController {

    private final TotpService totpService;

    @Operation(summary = "生成2FA绑定信息")
    @PostMapping("/setup")
    public Result<TotpSetupResponse> setup() {
        Long userId = SecurityUtil.getCurrentUserId();
        String username = SecurityUtil.getCurrentUsername();
        return Result.success(totpService.generateSetup(userId, username));
    }

    @Operation(summary = "验证并启用2FA")
    @PostMapping("/verify")
    public Result<Void> verify(@Valid @RequestBody TotpVerifyRequest request) {
        Long userId = SecurityUtil.getCurrentUserId();
        totpService.verifyAndEnable(userId, request.getCode());
        return Result.success();
    }

    @Operation(summary = "禁用2FA")
    @DeleteMapping
    public Result<Void> disable() {
        Long userId = SecurityUtil.getCurrentUserId();
        totpService.disable(userId);
        return Result.success();
    }

    @Operation(summary = "查询2FA状态")
    @GetMapping("/status")
    public Result<Map<String, Boolean>> status() {
        Long userId = SecurityUtil.getCurrentUserId();
        boolean enabled = totpService.isEnabled(userId);
        return Result.success(Map.of("enabled", enabled));
    }
}
```

          <img :src="qrCodeUrl" alt="2FA QR Code" width="200" height="200" />
        </div>
        <p class="secret-key">手动输入密钥：<code>{{ setupData?.secretKey }}</code></p>
        <el-button type="primary" @click="setupStep = 1">下一步</el-button>
      </div>

      <!-- Step 2: 验证码确认 -->
      <div v-if="setupStep === 1" class="setup-step">
        <p>请输入认证器上显示的 6 位验证码：</p>
        <el-input v-model="verifyCode" placeholder="000000" maxlength="6" style="width: 200px" />
        <div style="margin-top: 16px">
          <el-button @click="setupStep = 0">上一步</el-button>
          <el-button type="primary" :loading="verifying" @click="handleVerify">验证并启用</el-button>
        </div>
      </div>

      <!-- Step 3: 备用码 -->
      <div v-if="setupStep === 2" class="setup-step">
        <el-alert type="warning" :closable="false" title="请妥善保存以下备用码，每个只能使用一次" style="margin-bottom: 16px" />
        <div class="backup-codes">
          <code v-for="code in setupData?.backupCodes" :key="code" class="backup-code">{{ code }}</code>
        </div>
        <el-button type="primary" @click="handleFinish">完成</el-button>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import { setupTotp, verifyTotp, disableTotp, getTotpStatus } from '@/api/modules/totp'
import type { TotpSetupResponse } from '@/api/modules/totp'

const enabled = ref(false)
const showSetup = ref(false)
const setupStep = ref(0)
const setupData = ref<TotpSetupResponse | null>(null)
const verifyCode = ref('')
const verifying = ref(false)
const qrCodeUrl = ref('')

onMounted(async () => {
  const res = await getTotpStatus()
  enabled.value = res.enabled
})

const handleSetup = async () => {
  const res = await setupTotp()
  setupData.value = res
  // 使用 Google Charts API 生成二维码图片
  qrCodeUrl.value = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(res.qrCodeUri)}`
  showSetup.value = true
  setupStep.value = 0
}

const handleVerify = async () => {
  if (!verifyCode.value || verifyCode.value.length < 6) {
    ElMessage.warning('请输入6位验证码')
    return
  }
  verifying.value = true
  try {
    await verifyTotp(verifyCode.value)
    ElMessage.success('双因素认证已启用')
    setupStep.value = 2
  } catch (e: any) {
    ElMessage.error(e.message || '验证失败')
  } finally {
    verifying.value = false
  }
}

const handleDisable = async () => {
  await disableTotp()
  enabled.value = false
  ElMessage.success('双因素认证已禁用')
}

const handleFinish = () => {
  showSetup.value = false
  enabled.value = true
  setupData.value = null
  verifyCode.value = ''
}
</script>
