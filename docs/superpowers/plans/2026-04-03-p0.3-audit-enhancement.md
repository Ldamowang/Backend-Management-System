在 `@Around` 切面方法中，增加数据变更记录逻辑：

```java
@Around("@annotation(log)")
public Object around(ProceedingJoinPoint point, Log log) throws Throwable {
    long startTime = System.currentTimeMillis();
    SysOperationLog opLog = new SysOperationLog();

    // 基本信息（保持不变）
    opLog.setModule(log.module());
    opLog.setOperation(log.operation());
    opLog.setUsername(SecurityUtil.getCurrentUsername());
    // ... 其他基本信息赋值保持不变

    // 新增: 设置 targetType
    if (!log.targetType().isEmpty()) {
        opLog.setTargetType(log.targetType());
    }

    // 新增: 从 URL 路径中提取 targetId
    HttpServletRequest request = ((ServletRequestAttributes) RequestContextHolder.getRequestAttributes()).getRequest();
    Long targetId = extractTargetId(request.getRequestURI());
    opLog.setTargetId(targetId);

    // 新增: PUT/DELETE 请求记录 before_data
    String method = request.getMethod();
    if (("PUT".equals(method) || "DELETE".equals(method)) && !log.targetType().isEmpty() && targetId != null) {
        try {
            Object beforeObj = queryTargetById(log.targetType(), targetId);
            if (beforeObj != null) {
                opLog.setBeforeData(objectMapper.writeValueAsString(beforeObj));
            }
        } catch (Exception ignored) {
            // 查询失败不影响业务
        }
    }

    Object result;
    try {
        result = point.proceed();
        opLog.setStatus(1);
        opLog.setResponseCode(200);

        // 新增: PUT 请求记录 after_data
        if ("PUT".equals(method) && !log.targetType().isEmpty() && targetId != null) {
            try {
                Object afterObj = queryTargetById(log.targetType(), targetId);
                if (afterObj != null) {
                    opLog.setAfterData(objectMapper.writeValueAsString(afterObj));
                }
            } catch (Exception ignored) {
            }
        }
    } catch (Exception e) {
        opLog.setStatus(0);
        opLog.setErrorMsg(e.getMessage() != null ? e.getMessage().substring(0, Math.min(e.getMessage().length(), 500)) : "");
        throw e;
    } finally {
        opLog.setDuration(System.currentTimeMillis() - startTime);
        operationLogMapper.insert(opLog);
    }
    return result;
}

/**
 * 从 URI 中提取末尾的数字 ID（如 /api/users/123 → 123）
 */
private Long extractTargetId(String uri) {
    if (uri == null) return null;
    String[] parts = uri.split("/");
    for (int i = parts.length - 1; i >= 0; i--) {
        try {
            return Long.parseLong(parts[i]);
        } catch (NumberFormatException ignored) {
        }
    }
    return null;
}

/**
 * 根据 targetType 和 id 查询目标对象
 */
private Object queryTargetById(String targetType, Long id) {
    // 使用 Map 映射 targetType 到对应的 Mapper
    // 简单实现：通过 MyBatis SqlSession 动态查询
    String statement = "com.iflytek.admin.modules.system.mapper.Sys"
            + capitalize(targetType) + "Mapper.selectById";
    try (var session = sqlSessionFactory.openSession()) {
        return session.selectOne(statement, id);
    } catch (Exception e) {
        return null;
    }
}

private String capitalize(String str) {
    if (str == null || str.isEmpty()) return str;
    return str.substring(0, 1).toUpperCase() + str.substring(1);
}
```

const diffVisible = ref(false)
const currentDiff = ref({ before: '', after: '' })

const showDiff = (row: OperationLog) => {
  currentDiff.value = {
    before: row.beforeData || '{}',
    after: row.afterData || '{}'
  }
  diffVisible.value = true
}

const formatJson = (str: string) => {
  try {
    return JSON.stringify(JSON.parse(str), null, 2)
  } catch {
    return str
